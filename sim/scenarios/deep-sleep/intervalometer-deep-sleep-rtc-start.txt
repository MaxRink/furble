# The first half of the battery-backed RTC power cycle.
#
# The modelled StickC Plus2 row wakes through its BM8563 alarm, so the
# intervalometer's timed shutdown is allowed and Platform::powerOffUntil()
# records the sleep interval against the modelled chip before the process
# exits. The second half is intervalometer-deep-sleep-rtc-resume.txt, which
# runs against the same preferences store.
seed hw_profile stickc-plus2
seed rtc_model backed
seed rtc_epoch 1767225600
seed fauxny true
seed autoconnect true
seed ivl_sleep true
seed ivl_sleep_thr 10
seed interval_count 2
seed interval_delay 16
seed interval_shutter 30
seed interval_wait 0
assert hw.rtc_model backed
assert hw.rtc_clock valid
assert time.source rtc
wait 5000
action intervalometer
# The first shot's 30 ms exposure is followed by this full delay. Keep the
# wait beyond that boundary so the production sleep decision is reached.
wait 20000
