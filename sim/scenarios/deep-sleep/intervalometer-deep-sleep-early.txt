# An early virtual wake must wait for the persisted shot deadline. The fixture
# uses a 30-second absolute wall-clock deadline so the connection gate cannot
# consume the deadline before the virtual-time assertion starts.
seed fauxny true
seed autoconnect true
seed ivl_sleep true
seed timed_wake true
seed resume_fixture early
seed interval_count 2
seed interval_shutter 30
# The connection handshake is a host-scheduled event, while `wait` advances
# virtual device time quickly. Wait for the real connection before starting
# the deadline check so startup latency cannot masquerade as an early wake.
assert-eventually-virtual 5000 control.state active
wait 2000
assert camera.shutter_presses 0
assert ui.interval_state shutter
wait 30000
assert camera.shutter_presses 1
exit
