# display is allowed to reach off, so the reported estimate is the standing cost
# of the 50 Hz accelerometer poll against screen-off-30s.json.

seed inactivity 1
seed display_off 1

seed imu_wake 3
seed imu_trigger true

report setup.json
wait 30000
report gesture-idle-30s.json
