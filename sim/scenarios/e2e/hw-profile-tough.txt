# M5Tough profile model. Tough shares Core2 but uses LDO3 for backlight.
seed hw_profile tough
wait 100
assert hw.profile tough
assert hw.mcu esp32
assert hw.flash_kb 16384
assert hw.pmic axp192
assert hw.display 320x240
assert hw.touch yes
assert hw.sd yes
# The model reports the firmware's own feedbackSound column. The Tough row sets
# it, M5Unified's board_M5Tough speaker setup configures the Core2 codec path with
# a higher magnification for the waterproof case, and
# tests/host/board_profile_test.cpp asserts the NS4168 1 W speaker. The branch's
# claim that this board has no sound was wrong.
assert hw.audio yes
assert hw.rtc yes
# The Tough's BM8563 carries the BT1 backup cell and its alarm is the wake path
# (plans/354), so its row is backed.
assert hw.rtc_backed yes
assert mem.internal_total 532480
assert mem.component.lvgl_draw 20480
assert hw.usb uart
assert hw.wifi yes
assert hw.ethernet no
assert hw.ble yes
assert hw.gauge voltage_current
assert hw.psram quad
assert mem.psram_total 8388608
assert hw.ir no
assert pmic.watchdog unavailable
assert mem.component.main_stack 16384
exit
