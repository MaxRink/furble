assert hw.sd yes
assert hw.audio yes

assert mem.component.lvgl_draw 20480
assert hw.usb uart
assert hw.wifi yes
assert hw.ethernet no
assert hw.ble yes
assert hw.gauge voltage_current
assert hw.psram quad
assert mem.psram_total 8388608
assert hw.ir yes
assert pmic.watchdog unavailable
