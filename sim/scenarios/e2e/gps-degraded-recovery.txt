# Drive the production GPS duty-cycle wake timeout into degraded retry by
# suppressing only the fake UART event source, then restore a valid receiver.
seed gps true
seed gps_rate 1
seed gps_power 1
seed gps_duty 5
seed gps_uart_mode ack
wait 4000
assert gps.state standby
uart-mode pause
wait 12000
assert gps.degraded 1
assert gps.degraded_retries 1
assert power.no_light_sleep 0
wait 30000
assert power.no_light_sleep 0
uart-mode ack
# assert-eventually only yields wall time. Advance the simulator clock past the
# second bounded retry deadline so the production GPS task can run its probe.
wait 6000
assert-eventually 5000 gps.degraded 0
assert-eventually 5000 power.no_light_sleep 0
# The healthy duty-cycle scheduler may legitimately reacquire the lock on the
# next predicted window. The eventual assertion above proves recovery released
# the degraded lock without racing that subsequent healthy cycle.
exit
