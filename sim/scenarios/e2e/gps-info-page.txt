# Exercise the existing Settings > GPS > GPS Data page with the simulator's
# deterministic fake-UART fix. The injected NMEA burst renders 48.11730 N,
# 11.51667 E, 8 satellites and 42.0 km/h.
seed gps true
wait 300
assert ui.page main
action nav gps_data
wait 300
assert ui.page gps_data
wait 2000
assert ui.gps_lat 48.11730
assert ui.gps_lon 11.51667
assert ui.gps_satellites 8
assert ui.gps_fix yes
assert ui.gps_speed 42.0
assert ui.overflow no
exit
