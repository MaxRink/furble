# Board row versus hardware model: M5Stack Core2.
#
# AXP192 with DCDC3 backlight, LDO3 vibration and LDOIO0 Grove 5 V, the BM8563
# with its rechargeable cell, a speaker, a touch panel and an SD slot, and no
# IR emitter. See board-row-stickc.txt for the assertion contract.
seed hw_profile core2
wait 100

assert hw.profile core2
assert board.name Core2
assert hw.flash_kb 16384

assert hw.sd yes
assert board.sd yes

assert hw.audio yes
assert board.audio yes

assert hw.rtc yes
assert hw.rtc_backed yes
assert board.rtc_backed yes

# Timed wake path.
assert hw.timed_wake none
assert board.timed_wake none

assert hw.ir no
assert board.ir no

assert hw.touch yes

assert board.battery_mah 500
assert platform.battery.capacity 500

assert hw.gauge voltage_current
assert platform.battery.caps.level yes
assert platform.battery.caps.charging yes
assert board.battery_charging yes
assert platform.battery.caps.voltage yes
assert board.battery_voltage yes
assert platform.battery.caps.current yes

assert power.off_path axp192-off
assert power.can_off yes
assert board.can_power_off yes

assert board.gps_rail yes
assert power.rail.gps off
# The Port A rail and the external 5 V net are one switch on this board.
assert power.rail.ext5v off
assert power.rail.ext5v_drops 0
exit
