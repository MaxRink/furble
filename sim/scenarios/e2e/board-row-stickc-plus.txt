# Board row versus hardware model: M5StickC Plus.
#
# See board-row-stickc.txt for the two-sided assertion contract. The Plus adds
# the buzzer the firmware drives for feedback sound; the RTC and the AXP192
# rails are the same as the StickC.
seed hw_profile stickc-plus
wait 100

assert hw.profile stickc-plus
assert board.name StickC+

assert hw.sd no
assert board.sd no

assert hw.audio yes
assert board.audio yes

assert hw.rtc yes
assert hw.rtc_backed yes
assert board.rtc_backed yes

# Timed wake path.
assert hw.timed_wake rtc-alarm
assert board.timed_wake rtc-alarm


assert hw.ir yes
assert board.ir yes

assert hw.touch no

assert board.battery_mah 120
assert platform.battery.capacity 120

assert hw.gauge voltage_current
assert platform.battery.caps.charging yes
assert board.battery_charging yes

assert power.off_path axp192-off
assert power.can_off yes
assert hw.can_power_off yes
assert board.can_power_off yes

assert board.gps_rail yes
assert power.rail.ext5v n/a
# The remaining row columns, each pinned on both sides.
# The model twin of the row's power-off path is power.can_off, above.
assert board.power_off_path yes
assert hw.rtc_on_bus no
assert board.rtc_on_bus no
assert hw.led_active_low yes
assert board.led_active_low yes
assert hw.ir_mark_low yes
assert board.ir_mark_low yes
assert hw.gps_pin_source board-support
assert board.gps_pin_source board-support
assert hw.gps_rx_pin -1
assert board.gps_rx_pin -1
assert hw.gps_tx_pin -1
assert board.gps_tx_pin -1
exit
