# A configured Shutter of 0 must release the shutter on the next UI pass, not at
# whatever period the timer carried over from the previous state. A 15 s Delay
# tells the two outcomes apart immediately. The first frame cannot show the
# defect because the timer still carries its 100 ms creation period; the second
# frame is where a 0 Shutter used to hold for the Delay.
seed fauxny true
seed interval_count 2
seed interval_delay 15
seed interval_shutter 0
seed interval_wait 0
wait 300
action connect
wait 3000
assert-eventually-virtual 3000 control.state active
assert-eventually-virtual 2000 ui.connected yes

action nav timer
wait 300
assert ui.page timer
action intervalometer
wait 500
assert ui.page timer_run

# First frame: the press and its release both land well inside one second.
assert-eventually-virtual 1000 camera.shutter_presses 1
assert-eventually-virtual 1000 camera.shutter_releases 1

# Second frame: the Delay still separates the frames, but the release that
# follows the second press must not wait for it.
advance 15000
assert-eventually-virtual 1000 camera.shutter_presses 2
assert-eventually-virtual 500 camera.shutter_releases 2

exit
