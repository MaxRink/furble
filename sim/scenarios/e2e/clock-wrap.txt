# Keep the PM1 deadline on the far side of the uint32_t wrap. The first assert
# catches an absolute comparison that treats a wrapped deadline as already due.
seed clock_ms 4294962295
seed watchdog true
assert platform.watchdog armed

# A single stall avoids the normal update path feeding the watchdog between
# intermediate assertions. The 45 second PM1 deadline is crossed after the
# uint32 wrap.
stall 45001
# M5PM1.h REG_WDT_CNT 0x0A resets the system at zero, so crossing the deadline is
# a PMIC-driven power cycle and not a status flag.
assert platform.watchdog cycled
assert pmic.watchdog_cycles 1
exit
