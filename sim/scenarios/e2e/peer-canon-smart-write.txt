# Canon Smart write path: shutter and split AF bytes on the mode service.
#
# The Canon Smart peer journals the 00010030-style shutter characteristic. A
# press is 00 01, a release 00 02, AF on 00 03 and AF off 00 04, all written
# without a response. The connect completes its pairing name, identity and mode
# writes before the journal baseline. Evidence: lib/furble/CanonEOSSmart.cpp
# and the CS-16 to CS-27 branch of CanonSmartVirtualCamera.
seed ble_peers canon-smart
seed ble_saved true
seed ble_client_selfdelete true
wait 300
assert ble.peers 1
assert ble.saved_cameras 1

# Pairing name, client UUID, client name, client type, identity commit and mode
# select.
action connect
assert-eventually-virtual 20000 control.state active
assert control.connected 1
assert-eventually-virtual 3000 ui.connected yes
assert ui.capability_focus yes
assert ble.peer.0.writes 6

action shutter
wait 200
assert-eventually-virtual 1000 camera.shutter_presses 1
assert-eventually-virtual 1000 camera.shutter_releases 1
assert camera.focus_presses 0
assert-eventually-virtual 1000 ble.peer.0.writes 8
assert ble.peer.0.writes 8
assert ble.peer.0.last_write 0002

action button-mode one-button
action main-press-hold
wait 200
assert-eventually-virtual 1000 camera.focus_presses 1
assert-eventually-virtual 1000 camera.focus_releases 1
assert-eventually-virtual 1000 ble.peer.0.writes 10
assert ble.peer.0.writes 10
assert ble.peer.0.last_write 0004

action disconnect
assert-eventually-virtual 5000 control.state idle
assert control.connected 0
assert control.targets 0
assert-eventually-virtual 3000 control.zombies 0
assert-eventually-virtual 3000 ble.live_clients 0
assert-eventually-virtual 3000 ui.connected no
exit
