# The virtual radio re-advertises the same peer every 100 ms for the whole
# scan, so the duplicate-address guard in CameraList::match is exercised for
# real: a repeated advertisement must not create a second camera row.
seed fauxny false
seed ble_peers fuji
seed scan_timeout 2
wait 300

home
wait 200
action nav scan
wait 1000
assert ui.page scan
assert camera.count 1
wait 1500
assert camera.count 1
assert scan.end_callbacks 1

# Return to the menu and scan again. The peer keeps advertising, but the camera
# list must remain deduplicated.
home
wait 200
action nav scan
wait 1000
assert ui.page scan
assert camera.count 1
wait 1500
assert camera.count 1
assert scan.end_callbacks 2
exit
