# Gesture controls follow the live IMU, including after the sensor disappears
# while the Sensors page is open. Complete rows stay readable and reachable.
seed imu true
seed imu_wake 1
seed imu_trigger true
wait 400
action nav sensors
wait 200
assert ui.page sensors
# On the physical 80x160 layout the complete Sensors rows scroll by 7px. Keep
# every label whole and both ends reachable while this scenario exercises the
# live gesture gates. Separate touch-only companions retain the strict fit.
assert ui.label_overlaps 0
assert ui.cut_labels 0
assert ui.scroll_top 0
action scroll bottom
wait 100
assert ui.scroll_bottom 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
action scroll top
wait 100
assert ui.scroll_top 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
assert ui.imu_gesture_controls_enabled yes
assert ui.gesture_timer yes

action nav gestures
wait 200
assert ui.page gestures
action scroll bottom
assert ui.scroll_bottom 0
action scroll top
assert ui.scroll_top 0
back
wait 200

# Losing the sensor disables every gesture control and stops the poll timer, so
# a failed IMU cannot burn the 50 Hz timer forever.
action imu.disable
wait 200
assert ui.imu_gesture_controls_enabled no
assert ui.gesture_timer no
exit
