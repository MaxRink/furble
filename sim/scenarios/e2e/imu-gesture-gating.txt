# Gesture controls follow live IMU availability, including the wake roller.
# This models a sensor becoming unavailable while the Sensors page is open.
seed imu true
seed imu_wake 1
seed imu_trigger true
wait 400
action nav sensors
wait 100
assert ui.imu_gesture_controls_enabled yes

action imu.disable
wait 100
assert ui.imu_gesture_controls_enabled no
exit
