# Pre-#392 WebUI supervisor allocation shape.
#
# Before #392 the supervisor task was 8 KiB and had no heap gate, so on the
# StickS3 it failed inside xTaskCreate and the device boot-looped. The gate is
# deliberately disabled here to model the old branch: the 8 KiB request must
# fail with the hardware-shaped "Mem alloc fail" line, the firmware must log
# that it could not create the supervisor, and the UI must stay alive with a
# fed watchdog.
seed heap_internal_free 6800
seed heap_internal_largest 2048
seed wifi_connected true
seed web_ui true
seed webui_password true
seed webui_gate off
wait 100

assert heap.active yes
assert webui.supervisor alloc_failed
assert webui.server off
assert webui.gate supervisor_task
assert webui.fail_closed yes
assert heap.alloc_failures 1
assert heap.last_failed_task webui_supervisor
assert heap.free 6800
assert heap.largest 2048
assert ui.page main
assert platform.watchdog armed

# The 12 KiB HTTPS allocation was the second half of the hardware symptom.
# Nothing retries it without a supervisor, so the pool and failure count hold.
wait 15000
assert heap.alloc_failures 1
assert heap.free 6800
assert ui.page main
assert platform.watchdog armed

exit
