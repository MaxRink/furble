# The 2026-08-28 hardware incident left the Connected screen up while NEITHER
# camera had a live link. This scenario drives it through the production stack:
# two saved cameras connect, both links die at the transport, and the UI must
# leave the Connected presentation within a bound and show the reconnect
# indication instead of a false Connected screen. The driver-level liveness
# invariant is enforced throughout (no opt-out).
#
# Unlike false-connected-both-links-dead.txt the radio is left healthy here, so
# this also covers the recovery the control task performs on its own.
seed ble_peers fuji-pair
seed ble_saved true
seed reconnect true
wait 300
action connect-two
wait 3200
assert-eventually-virtual 3000 control.state active
assert control.targets 2
assert control.connected 2
assert-eventually-virtual 2000 ui.connected yes
assert ui.reconnecting no
assert ui.status_text Connected
# Both links die at once, mirroring the two-camera incident. The radio stays
# healthy, so the control task reconnects on its own.
action drop
wait 4000
assert control.state active
assert control.connected 2
assert ui.connected yes
assert ui.reconnecting no
assert ui.status_text Connected
assert ui.reconnect_count 0/2
# The enforced liveness invariant never fired across the whole flow.
assert ui.liveness_violations 0
exit
