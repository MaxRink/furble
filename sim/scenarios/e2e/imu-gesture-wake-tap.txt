# Wake Gesture set to Tap: a tap wakes the display, a shake does not.
seed imu true
seed imu_wake 1
seed inactivity 1
seed display_off 1
wait 31000
assert ui.display_state off
assert ui.gesture_events 0

# Shake: sustained deviation over the three-sample run.
action imu.accel 0 0 1.8
wait 120
action imu.accel 0 0 1
wait 120
assert ui.gesture_events 1
assert ui.gesture_last shake
assert ui.display_state off

# Clear the refractory period left by the gesture above.
wait 900

# Tap: one released impulse.
action imu.accel 0 0 2.6
wait 20
action imu.accel 0 0 1
wait 120
assert ui.gesture_events 2
assert ui.gesture_last tap
assert ui.display_state on

# The wake must reset the inactivity counter, or the display dims again at once.
wait 5000
assert ui.display_state on
exit
