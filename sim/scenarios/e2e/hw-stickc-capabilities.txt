# M5StickC capability facts, pinned one row at a time.
#
# The StickC is the board the firmware has the least margin on: no buzzer and no
# speaker, a 95 mAh cell, a GPIO35 IMU interrupt, and no timed wake at all.
# Every value here comes from the board row in src/FurbleBoard.cpp, which is
# what production reads, so this scenario fails if the simulator invents a
# capability the firmware does not own.
#
# Sources: plans/implemented/23-feedback-outputs.md (no buzzer, red LED on G10,
# https://docs.m5stack.com/en/core/m5stickc), plans/implemented/20-imu-hw-motion.md
# (MPU6886 INT on GPIO35, shared with the BM8563), plans/implemented/139-persistent-time.md
# (BM8563 on a PMU-supplied rail, so the calendar is battery backed).
#
# Teeth, each one line: flip hw.audio back to a table copy and the audio assert
# fails; return true from armMotionWake for every build and motion_wake fails on
# the Plus2 twin; hard-code the 250 mAh StickS3 capacity and battery_mah fails;
# drop the profile-driven rtc_backed and the backup assert fails; set this row's
# irMarkLow to false and the IR block below fails on hw.ir_mark_low. The level
# assert beside it is pinned on its own by hw-plus2-shared-led-ir.txt, so a
# flipped polarity is caught there even with the boolean asserted first here.
seed hw_profile stickc
seed imu true
seed imu_chip mpu6886
seed hw_motion 2
wait 300

# Audio: the original StickC has neither a buzzer nor a speaker.
assert hw.audio no

# Battery: 95 mAh, from the vendor product page.
assert hw.battery_mah 95

# Motion wake: MPU6886 INT reaches GPIO35 directly.
assert hw.motion_wake direct
assert hw.motion_wake_pin 35
assert ui.motion_wake yes

# Timed wake: the AXP192 has no self-wake path.
assert hw.timed_wake none
assert platform.timed_wake no

# RTC: BM8563, backed while the cell is fitted, no countdown armed at boot.
assert hw.rtc yes
assert hw.rtc_backed yes
assert hw.rtc_irq_armed no
assert rtc.enabled yes

# IR and LED are separate pins on this board, so a blink cannot key the emitter.
# The emitter is driven from the pin's cathode side, so a mark pulls the pin low.
assert hw.ir_pin 9
assert hw.ir_mark_low yes
assert hw.ir_mark_level low
assert hw.led_pin 10
assert hw.led_active_low yes
assert hw.led_shared_with_ir no

# Separate pins, so a light pulse during a frame cannot key the emitter. This is
# the control for hw-plus2-shared-led-ir.txt.
action ir-frame
action feedback-light
assert ir.led_pulses 1
assert ir.led_conflicts 0

# Buttons: A on 37, B on 39, power on the AXP192 key, never a GPIO.
assert hw.button_a 37
assert hw.button_b 39
assert hw.button_pwr none
assert hw.button_pwr_source axp192-key

# Flash: the part and the image are both 4 MB, DIO, no PSRAM.
assert hw.flash_kb 4096
assert hw.build_flash_kb 4096
assert hw.build_flash_mode dio
assert hw.build_psram absent
assert hw.psram absent
assert mem.psram_total 0
exit
