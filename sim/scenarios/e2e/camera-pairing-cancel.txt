# Cancel on the pairing modal must reject (plan 66 / #63). This modal exists to
# protect a numeric comparison, so the one invariant that matters is that the
# Cancel button reaches NimBLE as a reject and tears the link down, not that it
# merely closes the modal. Rebinding Cancel to answerPairing(true) has to fail
# here on every panel.
seed ble_peers fuji
seed ble_saved true
wait 300

action connect
assert-eventually-virtual 8000 control.state active
assert control.connected 1

action camera-pair-request confirm 428913
wait 400
assert ui.modal open
assert ui.modal_count 1
assert ui.pairing_kind confirm
assert ui.pairing_code 428913

# No radio to come back through, so the drop this rejection causes is final and
# the link state is unambiguous.
action ble-connect-fail
action camera-pair-reject
wait 500
assert ui.modal closed
assert ui.modal_count 0
assert ui.pairing_pending no
assert ble.pairing_answers 1
assert ble.pairing_answer reject

# The link the rejection tore down must end on the connect-failed path, not on
# a half-dismissed screen: no connected target, the progress box gone and the
# connected page not presented.
assert-eventually-virtual 12000 control.connected 0
assert-eventually-virtual 5000 ui.connect_box hidden
assert ui.connected no
exit
