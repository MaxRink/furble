# A fixed-baud receiver that never answers used to hold the GPS task in
# ACQUIRING forever: NO_LIGHT_SLEEP stayed held and the 5 V rail stayed on.
# The acquisition deadline must drop the rail, release the lock, and enter the
# normal absent state with its one retry.
seed gps true
seed gps_baud 9600
seed gps_receiver none
seed gps_uart_mode ack
wait 300
action nav gps_data
wait 300
assert ui.page gps_data

assert-eventually-virtual 40000 gps.receiver absent
assert power.no_light_sleep 0
wait 1500
assert-eventually 5000 ui.gps_cycle absent

# The single retry still recovers a receiver that appears later.
gps-receiver 9600
wait 30000
assert-eventually-virtual 50000 gps.receiver present
assert gps.baud 9600
wait 3000
assert-eventually 8000 gps.satellites 8
wait 1500
assert-eventually 5000 ui.gps_source uart
exit
