# A physically absent/failed sensor remains hidden even when the persisted IMU
# setting is enabled. This is a separate process because startup capability
# detection and the persisted setting are both initialization state.
seed fauxny true
seed imu true
seed imu_sensor false
wait 400
action nav settings
wait 150
assert ui.imu_button_visible no
assert ui.level_button_visible no
action nav diagnostics
wait 150
action nav imu
assert ui.page diagnostics
assert ui.imu_accel_x none
exit
