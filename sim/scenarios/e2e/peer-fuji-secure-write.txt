# Fujifilm Secure write path: the bytes the camera really receives.
#
# The Secure handshake adds the four-byte status acknowledgement and the 10 s
# geotag sync-interval write around the client name, so a saved connect
# journals three entries instead of the Basic two. Everything after that is the
# shared Fujifilm shutter path: a press is 01 00 then 02 00, a release is
# 01 00 then 00 00, and focus reuses the single-write half-press and release
# forms. Evidence: fujifilm-handlers.md sections 3.2, 3.3 and 4.
seed ble_peers fuji-secure
seed ble_saved true
seed ble_client_selfdelete true
wait 300
assert ble.peers 1
assert ble.saved_cameras 1

action connect
assert-eventually-virtual 20000 control.state active
assert control.connected 1
assert-eventually-virtual 3000 ui.connected yes
assert ble.peer.0.writes 3

action shutter
wait 200
assert-eventually-virtual 1000 camera.shutter_presses 1
assert-eventually-virtual 1000 camera.shutter_releases 1
assert camera.focus_presses 0
assert-eventually-virtual 1000 ble.peer.0.writes 7
assert ble.peer.0.writes 7
assert ble.peer.0.last_write 0000

action button-mode one-button
action main-press-hold
wait 200
assert-eventually-virtual 1000 camera.focus_presses 1
assert-eventually-virtual 1000 camera.focus_releases 1
assert-eventually-virtual 1000 ble.peer.0.writes 9
assert ble.peer.0.writes 9
assert ble.peer.0.last_write 0000

action disconnect
assert-eventually-virtual 5000 control.state idle
assert control.connected 0
assert control.targets 0
assert-eventually-virtual 3000 control.zombies 0
assert-eventually-virtual 3000 ble.live_clients 0
assert-eventually-virtual 3000 ui.connected no
exit
