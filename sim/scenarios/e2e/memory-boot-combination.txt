# WebUI + MQTT + companion at the pre-#394 StickS3 heap pressure.
#
# The pre-task MQTT owner gate requires 18,432 free / 12,288 largest. The
# 13,543 seed is below it, so the 10 KiB MQTT owner is never created. The WebUI
# supervisor still fits and the HTTPS server stays gated closed. The device
# must log both fail-closed decisions and keep the UI and watchdog alive.
#
# Requires the native MQTT simulator build: FURBLE_SIM_MQTT=1.
seed heap_internal_free 13543
seed heap_internal_largest 12788
seed wifi_connected true
seed web_ui true
seed webui_password true
wait 100

assert webui.supervisor created
assert webui.server blocked
assert webui.gate server_memory
assert webui.fail_closed yes
assert mqtt.task failed
assert mqtt.connected no
assert mqtt.connect_failures 0
assert heap.alloc_failures 0
assert heap.last_failed_task none
assert ui.page main
assert platform.watchdog armed

wait 5000
assert mqtt.task failed
assert mqtt.connected no
assert heap.alloc_failures 0
assert ui.page main
assert platform.watchdog armed

exit
