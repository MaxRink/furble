# TG1WDT reset record after WebUI memory pressure, not a panic loop.
#
# The reset that followed the WebUI pressure on hardware was a watchdog reset;
# the crash record must report task_wdt, not panic, and must be printed once.
# Every furble sdkconfig aborts on a failed allocation, so a recorded TG1WDT
# cannot follow an allocation failure: the production gate refuses the
# supervisor first and no allocation fails. The pre-gate allocation shape is
# the panic fixture sim/fixtures/hardware/panic-webui-pre-gate-alloc.txt.
seed hw_profile sticks3
seed heap_internal_free 6800
seed heap_internal_largest 2048
seed web_ui true
seed webui_password true
seed wifi_connected true
seed reset_reason task_wdt
wait 100

assert webui.supervisor blocked
assert webui.server off
assert webui.fail_closed yes
assert heap.alloc_failures 0
assert platform.reset_reason task_wdt
assert ui.page main

action crash-inject-tg1wdt
assert crash.pending yes

restart

assert crash.recovered yes
assert crash.reset task_wdt
assert crash.task sys_evt
assert platform.reset_reason task_wdt
assert ui.page main

restart

assert crash.recovered no
assert crash.reset none

exit
