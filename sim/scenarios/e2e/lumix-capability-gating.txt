# Capability gating: LUMIX has no focus opcode but supports a held exposure.
#
# `01` and `02` are Confirm and Start, never focus, so the focus control must be
# disabled. The shutter press/release pair can still hold a bulb exposure, so the
# Bulb row stays enabled.

seed ble_peers lumix
seed ble_saved true
wait 300
action toggle preset_picker
wait 100
action connect
wait 3200
assert ui.connected yes
assert ui.capability_focus no
assert ui.capability_bulb yes
assert ui.bulb_disabled no

# The right indicator is navigation on the connected menu, so the disabled
# state is asserted in shutter mode, where the focus control lives.
action blind
wait 100
assert ui.page shutter
assert ui.focus_disabled yes

# The same right indicator is the plus step on the Bulb Duration preset page.
# A capability refresh, which runs on every ACTIVE transition, must leave it
# enabled there: a reconnect while the page is open used to disable it again
# after configPresetControl() had restored it.
action nav bulb_duration
wait 200
assert ui.page bulb_duration
assert ui.focus_disabled no
# Fail the first reconnect so the UI observes a non-active state and resets
# its capability-refresh latch; then let the retry reach ACTIVE.
action capability-refresh
wait 100
assert ui.page bulb_duration
assert ui.focus_disabled no
exit
