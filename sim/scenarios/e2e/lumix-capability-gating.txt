# Capability gating: LUMIX has no focus opcode but supports a held exposure.
#
# `01` and `02` are Confirm and Start, never focus, so the focus control must be
# disabled. The shutter press/release pair can still hold a bulb exposure, so the
# Bulb row stays enabled.

seed ble_peers lumix
seed ble_saved true
wait 300
action connect
wait 3200
assert ui.connected yes
assert ui.capability_focus no
assert ui.capability_bulb yes
assert ui.bulb_disabled no

# The right indicator is navigation on the connected menu, so the disabled
# state is asserted in shutter mode, where the focus control lives.
action blind
wait 100
assert ui.page shutter
assert ui.focus_disabled yes
exit
