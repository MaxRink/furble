# Status-bar layout-stability and Bluetooth-icon-colour matrix.
#
# Track the remaining narrow-panel battery drift and guard the fixed Bluetooth
# icon colour during reconnects.
#
# ui.battery_drift is the battery icon x minus the anchor captured on its first
# read here, so 0 means it has not moved from its baseline position. The battery
# rides a left-packed status row, so it can slide as icons appear. The current
# matrix XPASSes on 135x240 and 320x240 but still reports drift on 80x160; keep
# these cross-icon-state checks as xassert until all three panels pass.
#
# ui.bt_icon reports the status-row Bluetooth icon: "hidden" when connected or
# idle, "red" when it marks a reconnect.
#
# Run on all three panel widths (80x160 / 135x240 / 320x240) from the CI job.
seed fauxny true
seed reconnect true
wait 400

# --- disconnected baseline: anchor the battery, BT icon absent -----------
# Icon set here (reconnect icon on, no reconnecting icon, no GPS icon) is the
# baseline every hard assert below compares against.
assert ui.battery_drift 0
assert ui.bt_icon hidden

# --- connected: same icon set, battery unmoved, BT icon absent -----------
action connect
wait 3200
assert-eventually-virtual 3000 control.state active
# A drop indicator is raised by the connected page's liveness poll, which runs
# every LIVENESS_POLL_PERIOD_MS (500 ms), so it legitimately appears up to that
# plus one 50 ms control tick after the link dies. The bound below is that
# budget with headroom, not a relaxation: a parked or dead timer never satisfies
# it. See plans/161-sim-real-control.md.
assert-eventually-virtual 2000 ui.connected yes
assert ui.battery_drift 0
assert ui.bt_icon hidden

# --- reconnecting: the reconnect BT icon appears -------------------------
action drop
wait 700
assert control.state connecting
assert-eventually-virtual 2000 ui.reconnecting yes
xassert board-varies ui.battery_drift 0
assert-eventually-virtual 2000 ui.bt_icon red
wait 2500
assert control.state active
# recovered: icon set is back to the baseline, so this holds on every panel.
assert ui.battery_drift 0
assert ui.bt_icon hidden

# --- GPS on: the GPS icon now also occupies the row ----------------------
action toggle gps
wait 400
xassert board-varies ui.battery_drift 0

# --- reconnecting with GPS on: two extra icons versus the baseline -------
action drop
wait 700
assert-eventually-virtual 2000 ui.reconnecting yes
xassert board-varies ui.battery_drift 0
assert-eventually-virtual 2000 ui.bt_icon red
wait 2500
assert ui.bt_icon hidden
# GPS icon still shown, so still off the baseline icon set on the narrow panel.
xassert board-varies ui.battery_drift 0
exit
