# IMU presence and read gating. The host seam starts from the scenario's IMU
# setting, so this proves disabled pages stay hidden and do not render stale
# values. Sensor presence and persisted opt-in are independent gates.
seed fauxny true
seed imu false
wait 400

# The main menu Level entry follows the same gate, and a nav onto the hidden
# button is a no-op rather than a reachable dead page.
assert ui.level_main_button_visible no
action expect unavailable nav level_main
wait 150
assert ui.page main

# The main menu open branch refreshes the gate every time the home page loads,
# so a runtime IMU loss hides the entry on the next return home.
action imu.disable
action nav settings
wait 150
home
wait 150
assert ui.page main
assert ui.level_main_button_visible no

action nav settings
wait 150
assert ui.imu_button_visible no
assert ui.level_button_visible no

action nav diagnostics
wait 150
assert ui.imu_button_visible no
action expect unavailable nav imu
wait 1200
assert ui.imu_accel_x none

action expect unavailable nav imu
wait 150
assert ui.page diagnostics
assert ui.imu_accel_x none
exit
