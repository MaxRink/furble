# MQTT under StickS3 heap pressure.
#
# The pre-task owner gate requires 18,432 free / 12,288 largest. The 13,543
# seed is below that floor, so the 10 KiB owner is never created. The service
# stays off and the device keeps the UI and watchdog alive.
#
# Run with FURBLE_SIM_MQTT=1 and an unreachable URI, for example
# FURBLE_SIM_MQTT_URI=mqtt://127.0.0.1:1.
seed heap_internal_free 13543
seed heap_internal_largest 12788
seed wifi_connected true
wait 100

assert mqtt.task failed
assert mqtt.connected no
assert mqtt.connect_failures 0
assert heap.free 13543
assert heap.alloc_failures 0
assert heap.last_failed_task none
assert ui.page main
assert platform.watchdog armed

wait 5000
assert mqtt.task failed
assert mqtt.connected no
assert heap.free 13543
assert ui.page main
assert platform.watchdog armed

exit
