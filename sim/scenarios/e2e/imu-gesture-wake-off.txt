# Wake Gesture off with Double-Tap Shutter on: the detector runs, so a gesture
# is still detected, but nothing wakes the display.
seed imu true
seed imu_wake 0
seed imu_trigger true
seed inactivity 1
seed display_off 1
wait 31000
assert ui.display_state off
assert ui.gesture_timer yes

action imu.accel 0 0 1.8
wait 120
action imu.accel 0 0 1
wait 120
assert ui.gesture_events 1
assert ui.display_state off

# MotionSource also must not reset inactivity while Wake Gesture is off. Hold
# the sensor still through its quiet window, then move it long enough for the
# one-second motion poll to see the transition.
wait 61000
assert ui.motion_state stationary
action imu.accel 0 0 2
wait 2000
assert ui.motion_state moving
assert ui.display_state off
action imu.accel 0 0 1
wait 1000

wait 900
# The panel is off, so the UI loop sleeps until the next timer is due, up to
# 20 ms (plan 111, P-070), and each scripted step costs one pass. A 1 ms wait
# holds each tap across two passes, so the 20 ms gesture sampler sees it.
# Back-to-back actions can fall between two samples, and a 20 ms wait holds
# the spike for the three samples that the detector reads as a shake.
action imu.accel 0 0 2.6
wait 1
action imu.accel 0 0 1
wait 80
action imu.accel 0 0 2.6
wait 1
action imu.accel 0 0 1
wait 120
assert ui.gesture_last double_tap
assert ui.display_state off
exit
