# Nikon Remote write path: the 2083 key frame, paired through the real UI.
#
# A Nikon saved record carries the pairing identity the client generated, so a
# discovery-only seed cannot be persisted by ble_saved. This scenario pairs the
# advertised peer through the production scan row, which saves it on the first
# ACTIVE transition, then reconnects with the normal Connect action. A press is
# 02 02 and a release 02 00 on 2083; the Remote profile has no focus command, so
# the held gesture must send nothing. Evidence:
# analysis/nikon-b600-handlers.md, lib/furble/NikonRemote.h and the 2083 branch
# of NikonRemoteVirtualCamera.
seed ble_peers nikon-remote
seed ble_client_selfdelete true
wait 300
assert ble.peers 1
assert ble.saved_cameras 0

home
wait 200
action nav scan
wait 500
assert ui.page scan
assert-eventually 2000 camera.count 1

# Pairing connect: the staged pair writes are the whole journal.
action scan-row 0
assert-eventually-virtual 20000 control.state active
assert control.connected 1
assert-eventually-virtual 3000 ui.connected yes
assert-eventually-virtual 3000 ble.saved_cameras 1
assert ble.peer.0.writes 2

action disconnect
assert-eventually-virtual 5000 control.state idle
assert-eventually-virtual 3000 ble.live_clients 0

# Saved reconnect from the camera list: the profile re-runs its pair writes.
action connect
assert-eventually-virtual 20000 control.state active
assert control.connected 1
assert ui.capability_focus no
assert ble.peer.0.writes 4

action shutter
wait 200
assert-eventually-virtual 1000 camera.shutter_presses 1
assert-eventually-virtual 1000 camera.shutter_releases 1
assert camera.focus_presses 0
assert-eventually-virtual 1000 ble.peer.0.writes 6
assert ble.peer.0.writes 6
assert ble.peer.0.last_write 0200

# No focus primitive: the held gesture must not reach the peer.
action button-mode one-button
action main-press-hold
wait 200
assert camera.focus_presses 0
assert ble.peer.0.writes 6
assert ble.peer.0.last_write 0200

action disconnect
assert-eventually-virtual 5000 control.state idle
assert control.connected 0
assert control.targets 0
assert-eventually-virtual 3000 control.zombies 0
assert-eventually-virtual 3000 ble.live_clients 0
assert-eventually-virtual 3000 ui.connected no
exit
