# End-to-end connect flow. Boot with the FauxNY test camera available, drive the
# connect that the Scan and Connect buttons trigger, and confirm the UI and
# Control both reach the connected state.
seed fauxny true
wait 300
assert control.state idle
assert ui.page main
action connect
wait 200
# CMD_CONNECT is dequeued in STATE_IDLE and published as STATE_CONNECT; the
# control task moves to STATE_CONNECTING on its next housekeeping tick. Assert
# the first phase here so the check does not depend on that tick boundary.
assert control.state connect
assert ui.connect_box visible
wait 2800
assert-eventually-virtual 3000 control.state active
assert control.connected 1
assert-eventually-virtual 2000 ui.connected yes
exit
