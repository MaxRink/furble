# Plan 155: prove the driver-level liveness invariant has teeth. The link_lies
# seed plus the link-lies-kill action construct the exact divergence from the
# 2026-08-28 hardware incident: every camera link goes dead while the fake
# control state machine still claims an active session, so the UI keeps
# presenting the Connected screen indefinitely. With enforcement on this run
# would abort with LIVENESS INVARIANT FAILED; the liveness_check opt-out keeps
# CI green while the ui.liveness_violations counter proves the invariant fired.
# Mutation check: with the invariant disabled in sim/driver.cpp, the counter
# assert below must fail; see plans/155-sim-ui-liveness.md.
seed fauxny true
seed reconnect true
seed link_lies true
# Opt out of enforcement only: detection still runs and counts violations.
seed liveness_check false
# Shorten the grace period so the violation latches inside a short wait.
seed liveness_grace_ms 500
wait 300
action connect
wait 1200
assert control.state active
assert control.connected 1
assert ui.connected yes
assert ui.liveness_violations 0
# Kill the link truth behind the control state machine's back.
action link-lies-kill
wait 200
# The divergence the fake control cannot otherwise express: no link is up, yet
# control still claims an active session and the UI still shows Connected.
assert control.state active
assert control.connected 0
assert ui.connected yes
# Sit in the divergence past the grace period: the invariant fires, the counter
# records it, and the opted-out run keeps going.
wait 1000
assert_min ui.liveness_violations 1
# The false Connected presentation persists, untouched by the recording.
assert control.state active
assert ui.connected yes
exit
