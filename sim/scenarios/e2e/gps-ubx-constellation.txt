# Four major constellations on an M10 is GPS, Galileo, BeiDou B1C and GLONASS:
# B1I does not run beside GLONASS (MAX-M10S integration manual UBX-20053088
# 2.1.3). The mask without QZSS and SBAS turns both off. That set tops out at
# 4 Hz (data sheet Table 2), so a 200 ms rate is clamped to 250 ms.
seed gps true
seed gps_model m10
seed gps_m10_hot true
seed gps_gnss 15
seed gps_rate 3
wait 8000
assert gps.backend ubx
assert-eventually 3000 gps.ubx.verified 1
assert m10.naks 0
assert gps.ubx.gnss 15
assert gps.ubx.gnss_refused 0
assert m10.ram.1031001f 1
assert m10.ram.10310021 1
assert m10.ram.10310022 1
assert m10.ram.1031000d 0
assert m10.ram.1031000f 1
assert m10.ram.10310025 1
assert m10.ram.10310024 0
assert m10.ram.10310020 0
assert gps.ubx.rate_clamped 1
assert gps.ubx.rate_ms 250
assert m10.ram.30210001 250
# AssistNow Autonomous does not cover B1C; it is still asked for, and the
# receiver accepted it.
assert m10.ram.10230001 1
assert-eventually 5000 gps.source uart
exit
