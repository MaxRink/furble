# AssistNow Autonomous is opt-in. With the setting on, CFG-ANA-USE_ANA 1
# reaches RAM and BBR and the receiver accepts it.

assert m10.ram.10230001 1
assert m10.bbr.10230001 1
assert m10.naks 0
exit
