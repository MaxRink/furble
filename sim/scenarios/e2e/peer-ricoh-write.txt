# Ricoh GR IV write path: one accepted capture command, no focus write.
#
# A Ricoh capture is a single OperationRequest write, 01 01 (START, AF), after
# the Drive Mode write that selects the immediate flavor. There is no release
# write: the camera completes the capture itself. The class reports no focus
# capability, so the one-button focus gesture must send nothing. Evidence:
# ricoh-handlers.md sections 5.12, 8.1 and lib/furble/Ricoh.cpp.
seed ble_peers ricoh
seed ble_saved true
seed ble_client_selfdelete true
wait 300
assert ble.peers 1
assert ble.saved_cameras 1

# A saved connect writes the Paired Device Name once.
action connect
assert-eventually-virtual 20000 control.state active
assert control.connected 1
assert-eventually-virtual 3000 ui.connected yes
assert ui.capability_focus no
assert ble.peer.0.writes 1

# Drive Mode 00 then OperationRequest 01 01, two accepted writes.
action shutter
wait 200
assert-eventually-virtual 1000 camera.shutter_presses 1
assert-eventually-virtual 1000 camera.shutter_releases 1
assert camera.focus_presses 0
assert-eventually-virtual 1000 ble.peer.0.writes 3
assert ble.peer.0.writes 3
assert ble.peer.0.last_write 0101

# No focus primitive: the held gesture must not reach the peer.
action button-mode one-button
action main-press-hold
wait 200
assert camera.focus_presses 0
assert ble.peer.0.writes 3
assert ble.peer.0.last_write 0101

action disconnect
assert-eventually-virtual 5000 control.state idle
assert control.connected 0
assert control.targets 0
assert-eventually-virtual 3000 control.zombies 0
assert-eventually-virtual 3000 ble.live_clients 0
assert-eventually-virtual 3000 ui.connected no
exit
