# ESP32-S3 headless profile model.
#
# The SDL simulator still brings up a panel, so this scenario asserts the
# headless profile identity, memory and radios without asserting display state.
seed hw_profile headless-s3
wait 100
assert hw.profile headless-s3
assert hw.mcu esp32s3
assert hw.flash_kb 8192
assert hw.psram octal
assert hw.pmic none
assert hw.display none
assert hw.touch no
assert hw.battery absent
assert hw.wifi yes
assert hw.ethernet no
assert hw.ble yes
assert hw.ir no
assert hw.sd no
assert hw.audio no
assert pmic.watchdog unavailable
assert mem.internal_total 524288
assert mem.psram_total 8388608
assert mem.component.main_stack 8192
assert mem.component.lvgl_draw 0
assert hw.usb serial_jtag
assert hw.wifi yes
assert hw.ethernet no
assert hw.ble yes
assert hw.gauge none
exit
