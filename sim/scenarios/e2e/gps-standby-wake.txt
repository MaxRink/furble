# Timed standby must emit PCAS12, enter standby, then recover a fresh UART fix
# after the receiver's virtual wake interval.
#
# `gps.fresh_fixes_parsed` is session-cumulative: it already contains the
# acquisition fix that armed standby, and a second burst can legitimately be
# parsed while the receiver settles before the command, so a fixed total is not
# a contract. The parser is not double counting: processNmea() accepts at most
# one fix per burst through FixSequence. `gps.duty_wake_fixes_parsed` counts only
# fixes from a burst after STANDBY, which is the wake under test, and the UART
# journal is cleared at the baseline so any command seen later is this wake's.
seed gps true
seed gps_power 1
seed gps_duty 5
seed gps_rate 1
seed gps_uart_mode ack
# Sample before the earliest possible 5 s wake, so the standby baseline cannot
# race the first wake and the duty-wake counter is still zero. Leave enough
# time for a loaded host to finish acquisition and arm standby.
wait 4500
assert gps.source uart
assert gps.state standby
assert uart.last $PCAS12,5*1B
assert_min gps.fresh_fixes_parsed 1
assert gps.duty_wake_fixes_parsed 0
uart-dump
assert uart.standby_5s_commands 0

# One receiver interval later at least one wake burst must have produced a
# parser-accepted fix and a new PCAS12,5 command. Keep these as lower bounds:
# a loaded host can let more than one legitimate wake pass inside the virtual
# wait, and that must not turn into a timing failure.
wait 6000
assert_min gps.duty_wake_fixes_parsed 1
assert_min gps.fresh_fixes_parsed 2
assert_min uart.standby_5s_commands 1

# Stop the receiver and let any burst already in flight settle, then clear the
# command journal. Nothing may manufacture another fix or duty command from the
# cached UART source after quiescence. A burst in flight at the pause still
# completes its duty cycle and sends PCAS12 on the way back to standby; a slow
# sanitizer runner can take longer than a fixed second for that, so the journal
# is cleared only once the receiver is back in standby.
uart-mode pause
wait 1000
assert-eventually-virtual 15000 gps.state standby
uart-dump
wait 6000
assert_min gps.duty_wake_fixes_parsed 1
assert_min gps.fresh_fixes_parsed 2
assert uart.standby_5s_commands 0
exit
