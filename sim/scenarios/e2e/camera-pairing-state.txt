# Camera pairing state machine coverage (plan 66 / #63). The UI must keep the
# displayed code tied to one pending request, offer no Confirm on a prompt that
# has nothing to confirm, and close the modal on cancel. Malformed request
# syntax is rejected at the scenario boundary, so those cases live in
# sim/scenarios/invalid and in tests/host/sim_action_parser_test.cpp instead.
seed ble_peers fuji
seed ble_saved true
wait 300

action connect
assert-eventually-virtual 8000 control.state active
assert control.connected 1

# Passkey display has only Cancel. A duplicate request must not replace the
# code shown by the already-open modal.
action camera-pair-request display 654321
wait 400
assert ui.modal open
assert ui.pairing_kind display
assert ui.pairing_code 654321
action expect no-effect camera-pair-request confirm 111111
wait 400
assert ui.modal open
assert ui.pairing_kind display
assert ui.pairing_code 654321
# The duplicate is not just dropped, it is refused at the wire. Leaving the
# peer's second comparison unanswered would hold its SMP procedure open behind
# a code nobody is looking at.
assert ble.pairing_answers 1
assert ble.pairing_answer reject

# Passkey display offers no Confirm button, so an accept has nothing to click.
action expect no-effect camera-pair-accept
wait 200
assert ui.modal open
assert ui.pairing_kind display

# Cancelling a display prompt injects no further numeric-comparison answer,
# because there was never a comparison to answer, but it still drops the link.
# The count therefore stays where the refused duplicate left it.
action ble-connect-fail
action camera-pair-reject
wait 500
assert ui.modal closed
assert ui.pairing_pending no
assert ble.pairing_answers 1
assert ble.pairing_answer reject
assert-eventually-virtual 12000 control.connected 0

# Nothing pending: both answers stay valid no-ops instead of reopening a modal.
action expect no-effect camera-pair-reject
action expect no-effect camera-pair-expire
wait 200
assert ui.modal closed
assert ui.pairing_pending no
exit
