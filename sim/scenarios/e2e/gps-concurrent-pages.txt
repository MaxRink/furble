# The GPS task decodes fake UART data on a background host thread while these
# LVGL timers read the same receiver. Visit both pages repeatedly so every
# consumer goes through the coherent GPS status snapshot.
seed gps true
seed gps_nmea true
wait 500

action nav gps_data
wait 2200
assert ui.page gps_data
assert ui.gps_speed 42.0
assert ui.gps_lat 48.11730
assert ui.gps_lon 11.51667
assert ui.gps_satellites 8
assert ui.gps_fix yes

# Exercise the real settings callback while the background task is actively
# decoding. TSan must prove each disable/enable reset is atomic with respect to
# the task-owned UART and configuration state.
action toggle gps
wait 200
assert setting.gps 0
action toggle gps
wait 500
assert setting.gps 1

action nav nmea
wait 2200
assert ui.page nmea
assert ui.diag.nmea yes

action nav gps_data
wait 2200
assert ui.page gps_data
assert ui.gps_speed 42.0
assert ui.gps_lat 48.11730
assert ui.gps_lon 11.51667
assert ui.gps_satellites 8
assert ui.gps_fix yes
exit
