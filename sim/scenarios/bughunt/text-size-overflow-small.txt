# Content-overflow regression for the global Small text size (text_size 0).
#
# Companion to overflow-sweep.txt, which only exercises the default Normal font.
# The Text size setting scales the LVGL theme font. Small shrinks it, so no page
# should ever gain overflow it did not already have, but the smallest font is
# still walked here for symmetry with the Large sweep. This scenario seeds Small,
# then walks the same pages overflow-sweep walks and asserts the compact pages
# fit. Run it on the same panels overflow-sweep runs on: the 80x160 M5StickC and
# the 135x240 boards. Small lowers the theme font 12 -> 10 on the 80x160 StickC
# and 16 -> 12 on the 135x240 boards.
#
# The compact run pages carry an "assert ui.overflow no". Menu pages keep the
# selected font and their icons, and scroll vertically when complete wrapped
# rows exceed the viewport.
# The long settings lists (features, gps, bluetooth, about, power, diagnostics)
# hold more rows than a short panel can show, so they scroll by design and only
# print their overflow state.
seed text_size 0
seed fauxny true
seed imu true
seed gps true
# The IMU seed puts the home Level row in place so the home fit assert below
# covers the fullest home menu the setting allows.
wait 300
# The seeded Small size was loaded on boot.
assert setting.text_size 0
# Home menu: must fit on the narrow panels.
assert ui.page main
assert ui.overflow no
action page settings
wait 100
print ui.page
print ui.overflow
action nav display
wait 100
# Display page: short, must fit on the narrow panels.
assert ui.page display
# This page scrolls at this size by design: a page honours the text size the
# user chose and scrolls when the rows stop fitting rather than shrinking the
# face. Before plan 168 it reported a fit only because its container was
# pinned to the page height and stacked the rows on top of each other, which
# no fit query can see. What has to hold is that nothing overlaps.
assert ui.label_overlaps 0
action page settings
action nav sensors
wait 100
# Physical 80px keeps the selected font and scrolls its complete Sensors rows.
# The Gestures page also scrolls by design; both pages keep their ends reachable.
assert ui.page sensors
assert ui.label_overlaps 0
assert ui.cut_labels 0
assert ui.scroll_top 0
action scroll bottom
wait 100
assert ui.scroll_bottom 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
action scroll top
wait 100
assert ui.scroll_top 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
action nav gestures
wait 100
assert ui.page gestures
action scroll bottom
assert ui.scroll_bottom 0
action scroll top
assert ui.scroll_top 0
action page settings
action nav features
wait 100
print ui.page
print ui.overflow
action page settings
action nav gps
wait 100
print ui.page
print ui.overflow
action page settings
action nav sensors
wait 100
# Re-entering Sensors after other settings routes must preserve the same
# readable scroll contract.
assert ui.page sensors
assert ui.label_overlaps 0
assert ui.cut_labels 0
assert ui.scroll_top 0
action scroll bottom
wait 100
assert ui.scroll_bottom 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
action scroll top
wait 100
assert ui.scroll_top 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
# The page that carries the Motion Engine roller, next to the gesture settings.
# It is an intentional scroll page: the roller alone runs 103 px past the
# 135x240 panel even at Small, so a fit assertion here would be false. Assert
# both scroll ends stay reachable, which is what a roller that pushed content
# out of reach would break, and which is what hw-motion-text-size.txt asserts
# at Large.
action nav gestures
wait 100
assert ui.page gestures
action scroll bottom
assert ui.scroll_bottom 0
action scroll top
assert ui.scroll_top 0
action page settings
action nav timer
wait 100
# Timer settings list (Count, Delay, Shutter, Wait): must fit on the narrow panels.
assert ui.page timer
# This page scrolls at this size by design: a page honours the text size the
# user chose and scrolls when the rows stop fitting rather than shrinking the
# face. Before plan 168 it reported a fit only because its container was
# pinned to the page height and stacked the rows on top of each other, which
# no fit query can see. What has to hold is that nothing overlaps.
assert ui.label_overlaps 0
action page settings
action nav bluetooth
wait 100
print ui.page
print ui.overflow
action page settings
action nav about
wait 100
print ui.page
print ui.overflow
action page settings
action nav power
wait 100
print ui.page
print ui.overflow
action page settings
action nav diagnostics
wait 100
print ui.page
print ui.overflow
# Remote shutter page: compact, must fit on the narrow panels.
action blind
wait 200
assert ui.page shutter
assert ui.overflow no
# Connected page: every full row stays readable and both scroll ends are reachable.
action connect
wait 3200
assert ui.connected yes
assert ui.page connected
assert ui.label_overlaps 0
assert ui.cut_labels 0
assert ui.scroll_top 0
action scroll bottom
wait 100
assert ui.scroll_bottom 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
action scroll top
wait 100
assert ui.scroll_top 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
exit
