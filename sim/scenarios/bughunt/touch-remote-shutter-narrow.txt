# The touch Remote page keeps its full-size controls on every modeled panel.
# On a narrow panel they stack into the page's scroll area instead of sharing
# three columns that are too small for the controls and labels.
seed no_touch false
seed fauxny true
seed saved_camera true
wait 400

action connect
wait 3200
assert ui.connected yes

action page shutter
wait 300
assert ui.nav_layout touch
assert ui.page shutter
assert ui.control_mode shutter
assert ui.label_overlaps 0
assert ui.cut_labels 0
action scroll bottom
wait 150
assert ui.scroll_bottom 0
assert ui.label_overlaps 0
assert ui.cut_labels 0
action scroll top
wait 150
assert ui.scroll_top 0
assert ui.label_overlaps 0
assert ui.cut_labels 0

# Leave the page, then use the blind entry as a separate real route.
home
wait 250
assert ui.page connected
action blind
wait 300
assert ui.page shutter
assert ui.control_mode shutter
assert ui.label_overlaps 0
assert ui.cut_labels 0
exit
