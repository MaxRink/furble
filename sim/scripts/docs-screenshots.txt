# Docs walkthrough capture, default theme, M5StickS3 135x240.
# Navigation uses the deterministic scenario actions. Headless input injection
# (key/btn) does not drive LVGL menu activation, so pages are reached through
# the same real menu-button clicks the nav action dispatches.
#
# The capture harness (sim/scripts/docs-capture.sh) runs this with the optional
# capability env vars set (FURBLE_SIM_IR / FURBLE_SIM_FEEDBACK / FURBLE_SIM_SD)
# so the Infrared, Feedback and Storage submenus render, FURBLE_SIM_NO_TOUCH set
# so the Remote page uses the non-touch layout, and FURBLE_SIM_RIG=0 so the
# header shows the shipped title. The boot splash is captured separately by the
# harness through FURBLE_SIM_CAPTURE_SPLASH, before the LVGL UI takes over.
seed fauxny true
seed gps true
seed saved_camera true
wait 500

# Main menu. A saved camera is seeded so the Connect and Delete entries enable.
home
wait 300
capture main
print ui.page

# Connect, Scan and Delete list pages. Connect and Delete list the saved camera;
# Scan runs the fake scan that surfaces the FauxNY test camera.
action nav connect
wait 400
capture connect-list
print ui.page

home
wait 300
action nav scan
wait 700
capture scan-list
print ui.page

home
wait 300
action nav delete
wait 400
capture delete-list
print ui.page

# Settings root and submenus.

# Bulb duration page reached through the preset-step action, which opens the
# Bulb Duration page and snaps the stored duration onto the preset series.
home
wait 250
action preset-step-up
wait 400
capture bulb
print ui.page

# Companion pairing modal. Inject a pending pairing PIN so the accept/reject
# dialog the companion service raises renders without a rig peer. Reject it after
# the capture so the modal does not linger over the following pages.
home
wait 250
action companion-pair-request
wait 500
capture companion-pairing
print ui.modal
action companion-reject
wait 250

# Camera numeric-comparison prompt. The FauxNY seam injects a deterministic
# request on the active camera, so this captures the production modal without
# requiring a Ricoh body or a live BLE security callback.
home
wait 250
action connect
wait 300
action camera-pair-request confirm 428913
wait 500
capture camera-pairing
print ui.pairing_kind
action camera-pair-reject
wait 250

# Connect flow. action connect drives the real connect the camera button
# triggers. Capture the connecting overlay first, then the connected page.
home
wait 250
action connect
wait 300
capture connecting
print ui.page
print ui.connect_box
wait 1700
capture connected
print ui.page
print ui.connected

# Remote shutter page while connected, non-touch layout on this narrow board.
action blind
wait 500
capture remote
print ui.page

exit
