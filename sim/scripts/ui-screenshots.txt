# Capture the modeled M5StickS3 UI through the real action navigation paths.
# Headless key injection does not activate LVGL menu buttons reliably, so every
# page is opened by its canonical action name. Each capture has a page guard so
# a mislabeled frame cannot silently enter the documentation gallery.
seed fauxny true
seed gps true
seed saved_camera true
wait 500

home
wait 300
assert ui.page main
capture m5stick-s3-main

action nav settings
wait 350
assert ui.page settings
capture m5stick-s3-settings

action nav display
wait 350
assert ui.page display
capture m5stick-s3-settings-display

action nav features
wait 350
assert ui.page features
capture m5stick-s3-settings-features

action nav gps
wait 350
assert ui.page gps
capture m5stick-s3-settings-gps

action nav gps_data
wait 1200
assert ui.page gps_data
capture m5stick-s3-gps

action nav timer
wait 350
assert ui.page timer
capture m5stick-s3-settings-timer

action nav interval_count
wait 350
assert ui.page interval_count
capture m5stick-s3-settings-timer-count

action nav interval_delay
wait 350
assert ui.page interval_delay
capture m5stick-s3-settings-timer-delay

action nav interval_shutter
wait 350
assert ui.page interval_shutter
capture m5stick-s3-settings-timer-shutter

action nav interval_wait
wait 350
assert ui.page interval_wait
capture m5stick-s3-settings-timer-wait

action nav theme
wait 350
assert ui.page theme
capture m5stick-s3-settings-theme

action nav tx_power
wait 350
assert ui.page tx_power
capture m5stick-s3-settings-tx-power

action nav about
wait 350
assert ui.page about
capture m5stick-s3-settings-about

home
wait 300
action connect
wait 1800
assert-eventually-virtual 60000 ui.connected yes
assert ui.page connected
capture m5stick-s3-connected
exit
